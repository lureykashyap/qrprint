Supabase SQL Editor मा यो SQL चलाउनुहोस्

-- QR TO PRINT: table + RLS fix (safe to rerun)
-- IMPORTANT: Replace YOUR_ADMIN_EMAIL with the exact email used in Supabase Auth.
create extension if not exists pgcrypto;

create table if not exists public.print_sessions (
 id uuid primary key default gen_random_uuid(),
 admin_id uuid not null references auth.users(id) on delete cascade,
 status text not null default 'waiting',
 file_path text, file_name text, mime_type text, file_size bigint,
 paper_size text not null default 'A4',
 copies integer not null default 1,
 customer_name text, customer_phone text, customer_note text,
 printer_id uuid, uploaded_at timestamptz, approved_at timestamptz,
 printed_at timestamptz, expires_at timestamptz not null default (now() + interval '24 hours'),
 created_at timestamptz not null default now(), updated_at timestamptz not null default now()
);

-- Add columns safely when an older version of the table already exists.
alter table public.print_sessions add column if not exists copies integer not null default 1;
alter table public.print_sessions add column if not exists customer_name text;
alter table public.print_sessions add column if not exists customer_phone text;
alter table public.print_sessions add column if not exists customer_note text;
alter table public.print_sessions add column if not exists printer_id uuid;
alter table public.print_sessions add column if not exists uploaded_at timestamptz;
alter table public.print_sessions add column if not exists approved_at timestamptz;
alter table public.print_sessions add column if not exists printed_at timestamptz;
alter table public.print_sessions add column if not exists expires_at timestamptz not null default (now() + interval '24 hours');
alter table public.print_sessions add column if not exists updated_at timestamptz not null default now();
create index if not exists print_sessions_admin_created_idx on public.print_sessions(admin_id, created_at desc);
alter table public.print_sessions enable row level security;

-- Remove policies from previous versions to avoid duplicate policy errors.
drop policy if exists "Admin can manage print sessions" on public.print_sessions;
drop policy if exists "qr_admin_insert_sessions" on public.print_sessions;
drop policy if exists "QR visitors can read session" on public.print_sessions;
drop policy if exists "QR visitor can mark waiting session uploaded" on public.print_sessions;

-- Admin can manage only rows belonging to their own auth user and allowlisted email.
create policy "Admin can manage print sessions" on public.print_sessions
for all to authenticated
using (auth.uid() = admin_id and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_ADMIN_EMAIL'))
with check (auth.uid() = admin_id and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_ADMIN_EMAIL'));

-- Prototype QR customer access. For public production use a one-time token + Edge Function.
create policy "QR visitors can read session" on public.print_sessions
for select to anon, authenticated using (expires_at > now());
create policy "QR visitor can mark waiting session uploaded" on public.print_sessions
for update to anon, authenticated using (status = 'waiting' and expires_at > now())
with check (status = 'uploaded' and expires_at > now());

-- Private upload bucket and policies.
insert into storage.buckets (id,name,public,file_size_limit,allowed_mime_types)
values ('print-files','print-files',false,26214400,array['application/pdf','image/png','image/jpeg','image/webp','image/gif','application/msword','application/vnd.openxmlformats-officedocument.wordprocessingml.document'])
on conflict (id) do update set public=false,file_size_limit=excluded.file_size_limit,allowed_mime_types=excluded.allowed_mime_types;
drop policy if exists "QR upload to print-files" on storage.objects;
drop policy if exists "QR read print-files" on storage.objects;
drop policy if exists "Admin read print-files" on storage.objects;
drop policy if exists "Admin delete print-files" on storage.objects;
create policy "QR upload to print-files" on storage.objects for insert to anon, authenticated with check (bucket_id='print-files');
create policy "QR read print-files" on storage.objects for select to authenticated using (bucket_id='print-files' and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_ADMIN_EMAIL'));
create policy "Admin delete print-files" on storage.objects for delete to authenticated using (bucket_id='print-files' and lower(coalesce(auth.jwt()->>'email','')) = lower('YOUR_ADMIN_EMAIL'));

सुरक्षाका लागि SQL मा admin@example.com लाई वास्तविक admin email ले बदल्नुहोस्। यो starter setup हो; public upload policies खुला छन्, त्यसैले production मा one-time token र server-side validation भएको Supabase Edge Function प्रयोग गर्नु राम्रो हुन्छ।
